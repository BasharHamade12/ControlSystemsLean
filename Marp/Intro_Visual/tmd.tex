\tikzset{
  mass/.style  = {fill=orange!15, draw=orange!70!black, line width=1.4pt},
  mech/.style  = {draw=blue!60!black, line width=1.6pt},
  ground/.style = {draw=black!70, line width=1.8pt},
  hatch/.style = {draw=black!70, line width=0.9pt},
  brace/.style = {draw=blue!70!black, line width=1.2pt, decorate,
                   decoration={brace,amplitude=4pt,mirror}},
  bracetext/.style = {font=\small\sffamily\bfseries, color=blue!70!black},
  sub/.style   = {font=\small\sffamily, fill=white, fill opacity=0.95, text opacity=1,
                   rounded corners=2pt, inner sep=3pt},
  force/.style = {draw=red!75!black, line width=1.6pt, -{Latex[length=3mm,width=2.4mm]}},
}

%% Vertical dashpot from (x,ytop) to (x,ybot), piston plate at ypist.
\newcommand{\vdashpot}[4]{%
  \draw[mech] (#1,#2) -- (#1,#3);
  \draw[mech] (#1-0.25,#3) -- (#1+0.25,#3);
  \draw[mech] (#1-0.3,#3-0.05) -- (#1-0.3,#4);
  \draw[mech] (#1+0.3,#3-0.05) -- (#1+0.3,#4);
  \draw[mech] (#1-0.3,#4) -- (#1+0.3,#4);
}

\begin{tikzpicture}[x=1cm,y=1cm, line join=round]
  % Ground with hatching.
  \draw[ground] (-0.5,0) -- (5.0,0);
  \foreach \x in {-0.5,-0.2,0.1,0.4,0.7,1.0,1.3,1.6,1.9,2.2,2.5,2.8,3.1,3.4,3.7,4.0,4.3,4.6,4.9} {
    \draw[hatch] (\x,0) -- (\x-0.14,-0.22);
  }
  % Primary suspension: spring k1, damper c1.
  \draw[mech,decorate,decoration={zigzag,segment length=5pt,amplitude=3.5pt}]
    (1.6,0) -- (1.6,0.9);
  \node[sub] at (1.05,0.45) {$k_1$};
  \vdashpot{3.4}{0.9}{0.45}{0.0}
  \node[sub] at (3.95,0.45) {$c_1$};
  % Primary mass m1, driven by u(t); y(t) is its displacement.
  \draw[mass] (1.0,0.9) rectangle (4.0,2.3);
  \node[sub] at (2.5,1.6) {$m_1$};
  \draw[force] (1.2,3.2) -- (1.2,2.3);
  \node[sub] at (0.72,2.75) {$u(t)$};
  \draw[force] (4.35,2.1) -- (4.35,1.2);
  \node[sub] at (4.75,1.65) {$y(t)$};
  % TMD suspension: spring k2, damper c2.
  \draw[mech,decorate,decoration={zigzag,segment length=5pt,amplitude=3.5pt}]
    (2.2,2.3) -- (2.2,2.9);
  \node[sub] at (1.8,2.6) {$k_2$};
  \vdashpot{3.4}{2.9}{2.6}{2.3}
  \node[sub] at (3.85,2.6) {$c_2$};
  % TMD mass m2.
  \draw[mass] (1.9,2.9) rectangle (3.7,3.8);
  \node[sub] at (2.8,3.35) {$m_2$};
  % Braces: TMD on top, primary structure below.
  \draw[brace] (5.2,2.3) -- (5.2,3.9);
  \node[bracetext, anchor=west] at (5.45,3.1) {TMD};
  \draw[brace] (5.2,0.0) -- (5.2,2.3);
  \node[bracetext, anchor=west] at (5.45,1.15) {Primary structure};
\end{tikzpicture}
